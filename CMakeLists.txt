cmake_minimum_required(VERSION 3.12)
project(libserialport C)

# --- 1. Autotools Compatibility & Visibility Definitions ---
if(NOT WIN32)
    # Generate an empty config.h in the binary directory
    if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/config.h")
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config.h" "/* Generated by CMake */\n")
    endif()
    include_directories(${CMAKE_CURRENT_BINARY_DIR})
endif()

# Gather all implementation source files
set(SERIALPORT_SOURCES
    serialport.c
    timing.c
)

if(WIN32)
    list(APPEND SERIALPORT_SOURCES windows.c)
    set(WINDOWS_LIBS setupapi cfgmgr32)
elseif(APPLE)
    list(APPEND SERIALPORT_SOURCES darwin.c)
    find_library(FRAMEWORK_IOKIT IOKit)
    find_library(FRAMEWORK_COREFOUNDATION CoreFoundation)
    set(OS_LIBS ${FRAMEWORK_IOKIT} ${FRAMEWORK_COREFOUNDATION})
else()
    list(APPEND SERIALPORT_SOURCES linux.c)
endif()

# --- 2. Core Library Target ---
add_library(serialport ${SERIALPORT_SOURCES})

# Fix: Use root directory for headers, set as SYSTEM so <libserialport.h> works
target_include_directories(serialport SYSTEM PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

# Fix: Ensure visibility macro SP_PRIV is defined across all platforms
target_compile_definitions(serialport PUBLIC SP_PRIV=)

if(WIN32)
    target_link_libraries(serialport PRIVATE ${WINDOWS_LIBS})
    # Special definition required for MSVC builds
    target_compile_definitions(serialport PRIVATE LIBSERIALPORT_MSBUILD=1)
elseif(APPLE)
    target_link_libraries(serialport PRIVATE ${OS_LIBS})
endif()

# --- 3. Examples Build Configuration ---
if(IS_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/examples")
    add_executable(list_ports examples/list_ports.c)
    target_link_libraries(list_ports PRIVATE serialport)
    target_include_directories(list_ports PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
endif()
